\begin{formalstatement}[Depth zero in the base frame]\label{pn:P134}
For tied data, fix a chart $\beta$ at $i\le m$, a represented point
$p\sim_{i,\beta}x$ and an element $\kappa$ of its level group. For every
$P\in B$ write $k=\mathrm{rat}(b_P)^{-1}\kappa\,\mathrm{rat}(b_P)$.
Then $\mathrm{Cong}_{s(P),0}(k)$ and $\mathrm{Cong}_{s(P),0}(k^{-1})$
both hold.
\end{formalstatement}\nodestatus{P134}
