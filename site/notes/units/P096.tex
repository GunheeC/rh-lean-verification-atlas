\begin{formalstatement}[Equality of localized class lattices]\label{pn:P096}
Fix standard column data, a word $u$, a stage $i\le m$, a place $v$,
classes $c,c'$, and $\gamma\in\mathrm{GL}_{N_i}(E)$ relating $c,c'$
at $v$. For every height-one prime $Q$ of $\mathcal O_E$ whose
contraction differs from $v_F$, the localizations in $E^{N_i}$ satisfy
\[
                         (\gamma L_{c'})_Q=(L_c)_Q .
\]
\end{formalstatement}
