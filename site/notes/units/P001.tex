\begin{formalstatement}[Goal: The Riemann hypothesis in Mathlib's formulation]\label{pn:P001}
For every $s\in\mathbb C$, if $\zeta(s)=0$, $s\ne1$, and $s\ne-2(n+1)$ for every $n\in\{0,1,\ldots\}$, then $\Re s=1/2$.
\end{formalstatement}
